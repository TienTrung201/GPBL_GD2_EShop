
# Tạo RAM swap 10GB
sudo fallocate -l 10G /swapfile
sudo chmod 600 /swapfile
sudo mkswap /swapfile
sudo swapon /swapfile
echo '/swapfile none swap sw 0 0' | sudo tee -a /etc/fstab
sudo sysctl vm.swappiness=10
echo 'vm.swappiness=10' | sudo tee -a /etc/sysctl.conf

# Cài đặt LXQt

sudo apt install sddm  # Trình quản lý hiển thị cho LXQt
sudo systemctl enable sddm  # Kích hoạt SDDM làm mặc định

sudo apt update
sudo apt install vivaldi-stable

Bước 3: Cài đặt noVNC trên máy chủ
sudo apt install novnc websockify -y
novnc --vnc localhost:5901 --listen 6080
# Tạo file cấu hình dịch vụ noVNC
sudo nano /etc/systemd/system/novnc.service

# Thêm nội dung sau vào file cấu hình
[Unit]
Description=noVNC Service
After=network.target

[Service]
ExecStart=/snap/bin/novnc --vnc localhost:5901 --listen 6080
Restart=always
User=root

[Install]
WantedBy=multi-user.target

# Tải lại daemon của systemd
sudo systemctl daemon-reload

# Kích hoạt dịch vụ để khởi động cùng hệ thống
sudo systemctl enable novnc.service

# Khởi động dịch vụ
sudo systemctl start novnc.service
sudo systemctl restart novnc.service

# Kiểm tra trạng thái dịch vụ
sudo systemctl status novnc.service
